 Event Propogations :- 
 there are two types of Propogations
 1. bubbling : here we use false (this is default value)
 ex: document.getElementById('images').addEventListener('click', function(e){
    console.log("clicked inside ul")
   },)
    document.getElementById('stone').addEventListener('click', function(e){
    console.log("clicked inside stone")
   },false) 
   here firstly apply events inner elements and later outter elements

2. capturing : here we use true 
document.getElementById('images').addEventListener('click', function(e){
    console.log("clicked inside ul")
   },true)
    document.getElementById('stone').addEventListener('click', function(e){
    console.log("clicked inside stone")
   },true)
    here firstly apply events outter elements and later inner elements

    using e.stopPropagation():
    we stop Propogations
    ex:
    document.getElementById('stone').addEventListener('click', function(e){
    console.log("clicked inside stone")
    e.stopPropagation()
   },)
